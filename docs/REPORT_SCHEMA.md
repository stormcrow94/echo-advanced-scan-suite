# ECHO report JSON schema (`echo-report` v1.0)

Every run writes `report/report-<lang>.json` (when `--format json|both`). The structure is **stable within a
major `schema_version`**: fields may be added in 1.x, but never renamed or removed. Consumers should check
`schema == "echo-report"` and the major part of `schema_version`.

Only human-readable text changes with the language (`name`/`description` of findings generated by ECHO
itself, module `detail`). Keys, enums and values coming from the tools stay the same.

```jsonc
{
  "schema": "echo-report",
  "schema_version": "1.0",
  "generator": { "name": "ECHO Advanced Scan Suite", "version": "2026.11" },
  "language": "en",                       // "en" | "pt-BR"
  "report": {
    "title": "Reconnaissance & Exposure Assessment",
    "client": "Example Corp",             // "" when not given
    "author": "Blue Team",                // "" when not given
    "generated_at": "2026-11-25T12:21:00-03:00"
  },
  "scan": {
    "id": "recon-example.com-2026-11-25-120000",
    "target": "example.com",
    "profile": "standard",                // passive | standard | deep
    "status": "completed",                // running | completed | failed
    "started_at": "2026-11-25T12:00:00-03:00",
    "finished_at": "2026-11-25T12:20:28-03:00",
    "duration_seconds": 1228,
    "rate_limit": 20,                     // requests/s sent to the target
    "top_ports": 100,
    "error": null,
    "tool_version": "2026.11"
  },
  "modules": [
    { "name": "subdomains", "status": "ok", "seconds": 74,
      "detail": "10 subdomains", "detail_raw": "10 subdomínios (subfinder:9 findomain:8 crt.sh:7)" }
    // status: ok | empty | skipped | failed
  ],
  "summary": {
    "subdomains": 10, "resolved_hosts": 9, "open_ports": 14, "web_services": 10,
    "urls_historical": 120, "urls_crawled": 40, "js_files": 2, "js_endpoints": 4,
    "tls_certificates": 4, "hosts_all_ports_open": 1, "findings_total": 13,
    "severity": { "critical": 0, "high": 1, "medium": 5, "low": 1, "info": 6, "unknown": 0 },
    "risk_rating": "high"                 // highest severity present, or "none"
  },
  "findings": [
    {
      "id": "git-config",                 // nuclei template id, "nikto-<id>", "zap-<pluginid>" or "echo-tls-*"
      "source": "nuclei",                 // nuclei | nikto | zap | tlsx
      "severity": "medium",               // critical | high | medium | low | info | unknown
      "name": "Git Configuration - Detect",
      "target": "https://staging.example.com/.git/config",
      "description": "…",
      "remediation": "…",
      "references": ["https://…"],
      "tags": ["config", "git", "exposure"],
      "evidence": ["…"],
      "timestamp": "2026-11-25T12:20:00-03:00"   // may be null
    }
  ],
  "assets": {
    "subdomains": ["example.com", "www.example.com"],
    "dns":   [{ "host": "www.example.com", "a": ["192.0.2.11"], "aaaa": [], "cname": [] }],
    "ports": [{ "host": "www.example.com", "port": 443 }],
    "ports_all_open_hosts": ["cdn.example.com"],   // hosts answering on ≥ PORTS_ALLOPEN_PCT % of ports
    "web":   [{ "url": "https://www.example.com", "status_code": 200, "title": "…", "webserver": "nginx",
                "technologies": ["Nginx", "WordPress:6.6"], "ip": "192.0.2.50", "cdn": "", "location": "" }],
    "tls":   [{ "host": "example.com", "port": "443", "tls_version": "tls13", "subject_cn": "example.com",
                "subject_an": [], "issuer": ["Let's Encrypt"], "not_after": "2027-02-01T00:00:00Z",
                "expired": false, "self_signed": false, "mismatched": false, "untrusted": false, "flags": [] }]
  },
  "urls": { "historical": ["…"], "crawled": ["…"] },
  "javascript": { "files": ["…"], "endpoints": ["/api/v1/users"] }
}
```

Findings are sorted by severity (critical → unknown), then name and target.
